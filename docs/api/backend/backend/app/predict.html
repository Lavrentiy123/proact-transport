backend.app.predict

Прогноз для тика: клиент ml-core с circuit breaker и встроенное fallback-правило.

  • Основной путь — один батч POST {ML_URL}/v1/predict на все борта тика (таймаут ml_timeout_s).
  • ml_fail_threshold ошибок подряд → breaker открыт на ml_open_s секунд, затем одна пробная попытка (half-open).
  • Пока ml-core недоступен — правило 0.6·cur_dev + 8 (cur_dev восстановлен детектором прибытий, NaN → 0) с quality=fallback. Это аварийный режим: на размеченном test это правило даёт MAE 136.1 с (нулевой прогноз — 103.3 с, m_online по LOBO — 83.0 с), см. отчёт.

  3- Основной путь — один батч ``POST {ML_URL}/v1/predict`` на все борта тика (таймаут ``ml_timeout_s``).
  4- ``ml_fail_threshold`` ошибок подряд → breaker открыт на ``ml_open_s`` секунд, затем одна пробная

 24FALLBACK_VERSION = "fallback-rule-0.6cur+8"
 25RULE_A, RULE_B = 0.6, 8.0
 26RULE_MAE_TEST_S = 136.1            # MAE правила на labels_test с восстановленным cur_dev (отчёт, шаг 5)
 27LAPLACE_Q90 = math.log(5.0)        # квантиль 0.9 распределения Лапласа в единицах MAE
 28LATE_S = 120.0
 29Z_10_90 = 2.563
 30SIGMA_MIN_S = 20.0

 33def p_late(pred: float, q10: float, q90: float, threshold_s: float = LATE_S) -> float:
 34    """P(задержка > threshold) в нормальном приближении по квантилям q10–q90."""
 35    sigma = max((q90 - q10) / Z_10_90, SIGMA_MIN_S)
 36    z = (threshold_s - pred) / sigma
 37    return float(min(1.0, max(0.0, 0.5 * math.erfc(z / math.sqrt(2.0)))))

 56def rule_predict(tr_id: int, cur_dev_s: float | None) -> Result:
 57    """Fallback-правило ``0.6·cur_dev + 8``; интервал — Лаплас по MAE правила на test."""
 58    cur = cur_dev_s if cur_dev_s is not None and math.isfinite(cur_dev_s) else None
 59    base = cur if cur is not None else 0.0
 60    pred = RULE_A * base + RULE_B
 61    half = LAPLACE_Q90 * RULE_MAE_TEST_S
 62    q10, q90 = pred - half, pred + half
 63    if cur is None:
 64        code, conf = "low_data", 1.0
 65    else:
 66        d = abs(pred - base)
 67        code, conf = ("early", 1.0) if pred < -60 else ("accumulated", abs(base) / (abs(base) + d) if abs(base) + d else 1.0)
 68    return Result(tr_id, pred, q10, q90, p_late(pred, q10, q90), code, float(conf), {}, FALLBACK_VERSION,
 69                  "fallback-rule")

117        """Батч в ml-core. ``None`` — ml-core недоступен (breaker открыт, таймаут, ошибка ответа)."""
118        if not self.enabled or not rows or self.state == "open":
119            return None
120        if self._client is None:
121            self._client = httpx.AsyncClient(base_url=self.url, timeout=self.timeout_s, transport=self.transport)
122        self.calls += 1
123        t0 = time.perf_counter()
124        try:
125            req = PredictRequest(model=model, rows=rows)
126            r = await self._client.post("/v1/predict", content=req.model_dump_json(),
127                                        headers={"Content-Type": "application/json"})
128            r.raise_for_status()
129            resp = PredictResponse.model_validate_json(r.content)
130        except Exception as e:
131            self.errors += 1
132            self.failures += 1
133            self.last_ok = False
134            self.last_error = f"{type(e).__name__}: {e}"
135            if self.failures >= self.fail_threshold:
136                self.open_until = time.time() + self.open_s
137                log.warning("ml-core unavailable (%s), breaker open for %.0f s", self.last_error, self.open_s)
138            return None
139        self.last_latency_ms = (time.perf_counter() - t0) * 1000.0
140        if self.failures:
141            log.info("ml-core back after %d failures", self.failures)
142        self.failures = 0
143        self.last_ok = True
144        self.last_error = None
145        self.model_version = resp.model_version
146        src = "ml-core" if model == "online" else f"ml-core:{model}"
147        return [Result(x.tr_id, x.delay_pred_s, x.delay_q10_s, x.delay_q90_s, x.p_late, x.cause_code,
148                       x.cause_confidence, dict(x.group_contrib_s), resp.model_version, src) for x in resp.results]
