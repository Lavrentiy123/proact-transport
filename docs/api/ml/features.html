features

Пакет признаков «ПроАкт.Транспорт»: одни и те же признаки при обучении и на потоке.

Публичное API:

  • FEATURE_NAMES, FEATURES_VERSION — 34 признака в фиксированном порядке;
  • build_features() — признаки одной точки, build_features_batch() — набора точек;
  • StopDetector — детектор прибытий на плановые остановки;
  • OnlineVehicle — состояние борта на потоке (буфер 30 мин + детектор + признаки);
  • load_traffic(), load_schedule() — чтение и нормализация данных.

Анти-утечка: признаки для (tr_id, T) используют только телеметрию с t ≤ T и плановое расписание; факты прибытия пакет не читает.

 56def features_from_arrays(T: float, t: np.ndarray, lat: np.ndarray, lon: np.ndarray, spd: np.ndarray,
 57                         sched: ScheduleArrays, target_idx: int | None, target_plan_s: float,
 58                         arr_idx: np.ndarray, arr_t: np.ndarray, cur_dev_s: float | None = None,
 59                         inzone: np.ndarray | None = None) -> dict[str, float]:
 60    """Ядро расчёта признаков (общее для офлайна и онлайна).
 61
 62    Args:
 63        T: момент прогноза, секунды.
 64        t, lat, lon, spd: валидные фиксы борта, по возрастанию ``t``; используются только
 65            ``T − 30 мин < t ≤ T``.
 66        sched: плановое расписание борта.
 67        target_idx: позиция целевой остановки в ``sched`` (``None`` — неизвестна).
 68        target_plan_s: плановое время целевой остановки, секунды.
 69        arr_idx, arr_t: прибытия, известные к ``T`` (позиции в ``sched`` и времена), в порядке обнаружения.
 70        cur_dev_s: официальная подсказка; ``None`` — восстановить (``rdev_last``, затем
 71            ``plan_stop_nearest_dev_s``, иначе NaN).
 72        inzone: заранее посчитанные флаги зоны посадки для ``t`` (иначе считаются здесь).
 73
 74    Returns:
 75        ``dict`` ровно с ключами :data:`FEATURE_NAMES` в их порядке.
 76    """
 77    f = dict.fromkeys(FEATURE_NAMES, NAN)
 78    plan = sched.plan_s
 79    horizon = float(target_plan_s - T)
 80    f["horizon_s"] = horizon
 81    mod = (T % 86400.0) / 60.0
 82    f["tod_sin"] = math.sin(2 * math.pi * mod / 1440.0)
 83    f["tod_cos"] = math.cos(2 * math.pi * mod / 1440.0)
 84    f["hour"] = float(int(mod // 60))
 85
 86    # ---- расписание ----
 87    upto = int(np.searchsorted(plan, T, side="right"))
 88    f["n_stops_between"] = float(int(np.searchsorted(plan, target_plan_s, side="right")) - upto)
 89    if upto > 0:
 90        f["since_last_plan_s"] = T - float(plan[upto - 1])
 91
 92    # ---- восстановленные отклонения (детектор) ----
 93    last_idx = None
 94    if len(arr_idx):
 95        last_idx = int(arr_idx[-1])
 96        rdev = arr_t - plan[arr_idx]
 97        f["rdev_last"] = float(rdev[-1])
 98        f["rdev_age_s"] = T - float(arr_t[-1])
 99        for w, name in ((300.0, "rdev_trend5"), (600.0, "rdev_trend10")):
100            j = np.nonzero(arr_t <= T - w)[0]
101            if len(j):
102                f[name] = float(rdev[-1] - rdev[j[-1]])
103    n_pend, lb = pending_stops(plan, last_idx, T)
104    f["n_pending"] = float(n_pend)
105    f["lb_delay_s"] = lb
106
107    # ---- телеметрия (T − 30 мин, T] ----
108    lo = int(np.searchsorted(t, T - LOOKBACK_S, side="right"))
109    hi = int(np.searchsorted(t, T, side="right"))
110    t, lat, lon, spd = t[lo:hi], lat[lo:hi], lon[lo:hi], spd[lo:hi]
111    if inzone is not None:
112        inzone = inzone[lo:hi]
113    n = len(t)
114    if n:
115        f["tel_age_s"] = T - float(t[-1])
116        f["last_speed"] = float(spd[-1])
117        for w in WINDOWS_MIN:
118            s = int(np.searchsorted(t, T - 60.0 * w, side="right"))
119            if s < n:
120                sw = spd[s:]
121                f[f"v_mean_{w}m"] = float(sw.mean())
122                f[f"stop_ratio_{w}m"] = float((sw < STOP_SPEED_KMH).mean())
123                if n - s >= 2:
124                    f[f"disp_{w}m"] = float(haversine_m(lon[s], lat[s], lon[-1], lat[-1]))
125                if w == 5 and n - s >= 2:
126                    f["v_std_5m"] = float(sw.std(ddof=1))
127        # стоянки за 5 мин: в зоне остановок (посадка) и вне (затор)
128        s5 = int(np.searchsorted(t, T - 300.0, side="right"))
129        if s5 < n:
130            zone = inzone[s5:] if inzone is not None else in_zone_flags(lat[s5:], lon[s5:], sched)
131            dt = np.minimum(np.diff(t[s5:]), DWELL_INTERVAL_CAP_S)
132            stopped = spd[s5:-1] < STOP_SPEED_KMH if n - s5 >= 2 else np.zeros(0, dtype=bool)
133            f["stop_in_zone_s_5m"] = float(dt[stopped & zone[:-1]].sum())
134            f["stop_out_zone_s_5m"] = float(dt[stopped & ~zone[:-1]].sum())
135
136        if target_idx is not None:
137            f["dist_to_target_m"] = float(haversine_m(lon[-1], lat[-1], sched.lon[target_idx], sched.lat[target_idx]))
138            k0 = int(np.searchsorted(plan, T - ROUTE_CAND_BACK_S, side="left"))
139            if k0 <= target_idx:
140                dd = haversine_m(lon[-1], lat[-1], sched.lon[k0:target_idx + 1], sched.lat[k0:target_idx + 1])
141                j = k0 + int(np.argmin(dd))
142                route = float(dd[j - k0] + (sched.chain_cum_m[target_idx] - sched.chain_cum_m[j]))
143                f["route_dist_m"] = route
144                f["plan_stop_nearest_dev_s"] = T - float(plan[j])
145                v10 = f["v_mean_10m"]
146                v_eff = max(v10, V_EFF_MIN_KMH) if not math.isnan(v10) else V_EFF_DEFAULT_KMH
147                f["proj_delay_s"] = route / (v_eff / 3.6) - horizon
148
149    if cur_dev_s is None:

151    f["cur_dev_s"] = float(cur_dev_s) if cur_dev_s is not None else NAN
152    return f

137    def _wrong_way(self, k: int, t: float, lat: float, lon: float) -> bool:
138        """Движется ли борт против маршрута у остановки ``k`` (по своим фиксам до ``t`` включительно)."""
139        brg = self._brg[k]

142        for tj, laj, loj in reversed(self._hist):
143            if t - tj > HEADING_LOOKBACK_S:

146                mv = bearing_deg(loj, laj, lon, lat)

150    def _step(self, t: float, lat: float, lon: float) -> tuple[int | None, float]:
151        if not (math.isfinite(lat) and math.isfinite(lon)):
152            return None, math.nan
153        hist = self._hist
154        hist.append((t, lat, lon))
155        while hist and t - hist[0][0] > HEADING_LOOKBACK_S:
156            hist.pop(0)
157        plan = self._plan
158        # остановки, окно которых закончилось, — пропущены
159        while self.i < self._n and t > plan[self.i] + WINDOW_AFTER_S:
160            self._missed.append((self.i, t))
161            self._min.pop(self.i, None)
162            self.i += 1
163        if self.i >= self._n or t < plan[self.i] - WINDOW_BEFORE_S:
164            return None, math.nan
165        early, late = WINDOW_BEFORE_S, WINDOW_AFTER_S
166        if self._rec_t and t - self._rec_t[-1] <= REL_MAX_AGE_S:
167            # правдоподобное отклонение: от min(dev, 0) − REL_BEFORE до max(dev, 0) + REL_AFTER
168            dev = self._rec_t[-1] - plan[self._rec_idx[-1]]
169            early = min(early, REL_BEFORE_S - min(dev, 0.0))
170            late = min(late, REL_AFTER_S + max(dev, 0.0))
171        hit = None
172        for k in range(self.i, min(self.i + N_CANDIDATES, self._n)):
173            pk = plan[k]
174            if t < pk - early:
175                break  # план отсортирован: дальше окна ещё не начались
176            if t > pk + late:
177                continue
178            d = haversine_scalar_m(lon, lat, self._lon[k], self._lat[k])
179            st = self._min.get(k)
180            if self._term[k]:
181                # начало рейса: ждём отправления; st = (минимум расстояния, последний фикс у остановки)
182                if st is None or d < st[0]:
183                    self._min[k] = (d, t)
184                elif d <= st[0] + TERM_NEAR_TOL_M:
185                    self._min[k] = (st[0], t)
186                elif st[0] < PASS_RADIUS_M and d > st[0] + TERM_RECEDE_M:
187                    if self._wrong_way(k, t, lat, lon):
188                        self._min.pop(k)  # уехал не по маршруту (на отстой, разворот) — ждём отправления
189                        continue
190                    hit = (k, st[1])
191                    break
192                continue
193            if d < ARRIVE_RADIUS_M:
194                if self._wrong_way(k, t, lat, lon):
195                    continue
196                hit = (k, t)
197                break
198            if st is not None and st[0] < PASS_RADIUS_M and d > st[0] + PASS_RECEDE_M:
199                if self._wrong_way(k, t, lat, lon):
200                    self._min.pop(k)
201                    continue
202                hit = (k, st[1])
203                break
204            if st is None or d < st[0]:
205                self._min[k] = (d, t)
206        if hit is None:
207            return None, math.nan
208        k, t_arr = hit
209        for j in range(self.i, k):
210            self._missed.append((j, t))
211        self._rec_idx.append(k)
212        self._rec_t.append(t_arr)
213        self._rec_det.append(t)
214        self.i = k + 1
